use a2a_protocol_sdk::prelude::*;

agent_executor!(MyAgent, |ctx, q| async {
    let emit = EventEmitter::new(ctx, q);
    emit.status(TaskState::Working).await?;
    emit.artifact("hello", parts).await?;
    emit.status(TaskState::Completed).await?;
    Ok(())
});
