# a2a-rust — Official Rust SDK for the A2A (Agent-to-Agent) protocol
#
# Indexing policy: fully open to well-behaved web crawlers, with a small
# crawl-delay for low-effort bots to avoid hammering the GitHub Pages
# origin. AI training crawlers are allowed because this repository is
# Apache-2.0 licensed and we want the documentation to be discoverable
# and, yes, ingested. If you want to opt specific bots out, add
# `User-agent: X` stanzas above the catch-all.

User-agent: *
Allow: /

# mdbook's `print.html` aggregates every chapter into one huge page as a
# print-friendly alternative view. mdbook already emits a `<meta
# name="robots" content="noindex">` on it, but disallowing the URL here
# stops crawlers from wasting budget on it in the first place and
# eliminates the "Excluded by noindex tag" warning in Google Search
# Console. The page is still reachable by humans who click "Print this
# book" in the sidebar.
Disallow: /print.html

Crawl-delay: 1

# Explicitly surface the sitemap so Googlebot / Bingbot / DuckDuckBot
# discover new pages without waiting for a link crawl.
Sitemap: https://a2a-rust.com/sitemap.xml
